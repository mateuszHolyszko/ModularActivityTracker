// Different loading styles
LoadingWidget loading1, loading2, loading3, loading4;

loading1.setPosition(200, 200);
loading1.setColor(glm::vec4(1.0f, 0.3f, 0.3f, 1.0f)); // Red

loading2.setStyle(LoadingWidget::Style::DOTS_ROTATING);
loading2.setPosition(400, 200);

loading3.setStyle(LoadingWidget::Style::BAR);

loading3.setColor(glm::vec4(0.3f, 0.3f, 1.0f, 1.0f)); // Blue

loading4.setStyle(LoadingWidget::Style::SPINNER);
